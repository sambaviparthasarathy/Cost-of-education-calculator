const COACHING_DATA = {"PRIMARY": 1313, "MIDDLE": 2189, "SECONDARY": 4183, "HIGHER SECONDARY": 6384};


let currentArea = 'Rural';

let coachSelected = true;

let chart = null;


function populateStates() {

    const select = document.getElementById('stateSelect');

    Object.keys(MASTER_DATA).sort().forEach(state => {

        if(state === 'all-India') return;

        let opt = document.createElement('option');

        opt.value = state; opt.innerHTML = state;

        select.appendChild(opt);

    });

}


function setArea(area) {

    currentArea = a;

    document.getElementById('areaRural').className = (area === 'Rural' ? 'pill on active' : 'pill');

    document.getElementById('areaUrban').className = (area === 'Urban' ? 'pill on active' : 'pill');

    calculate();

}


function setCoaching(val) {

    coachSelected = val;

    document.getElementById('coachYes').className = (val ? 'pill on active' : 'pill');

    document.getElementById('coachNo').className = (!val ? 'pill on active' : 'pill');

    calculate();

}


function calculate() {

    const state = document.getElementById('stateSelect').value;

    const level = document.getElementById('levelSelect').value;

    const type = document.getElementById('typeSelect').value;


    const stateObj = MASTER_DATA[state] || MASTER_DATA['all-India'];

    const areaObj = stateObj[type] || MASTER_DATA['all-India'][type];

    

    // Fallback logic for missing data points

    const fee = areaObj[currentArea]?.[level] || areaObj['Urban']?.[level] || MASTER_DATA['all-India'][type]['Urban'][level];

    const coachFee = coachSelected ? (COACHING_DATA[level]) : 0;

    const total = fee + coachFee;


    document.getElementById('totalDisplay').innerHTML = "₹" + total.toLocaleString('en-IN');

    document.getElementById('breakdownText').innerHTML = `School Fees: ₹${fee.toLocaleString('en-IN')} | Coaching: ₹${coachFee.toLocaleString('en-IN')}`;

    document.getElementById('results').style.display = 'block';


    updateChart(total);

}


function updateChart(userVal) {

    const ctx = document.getElementById('myChart').getContext('2d');

    const indiaAvg = 13051; // National average across all types


    if (chart) chart.destroy();


    chart = new Chart(ctx, {

        type: 'bar',

        data: {

            labels: ['Your Selection', 'All-India Avg'],

            datasets: [{

                label: 'Annual Cost (₹)',

                data: [userVal, indiaAvg],

                backgroundColor: ['#7eb8f7', '#1a4fa0']

            }]

        },

        options: {

            responsive: true,

            maintainAspectRatio: false,

            scales: { y: { beginAtZero: true } }

        }

    });

}


window.onload = function() {

    populateStates();

    calculate();

};
